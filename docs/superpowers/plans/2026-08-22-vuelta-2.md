## El orden que hay que entender antes de escribir nada

Esta es la decisión de diseño de la vuelta, y si la lees por encima te va a
morder.

Las variables con retardo necesitan historia. Si construyes las variables
*después* de partir, la primera semana de noviembre no tiene de dónde sacar su
`lag_168h` y se queda sin datos. Así que el orden correcto es:

```
load_all()  ->  build_features()  ->  split()
```

Variables primero, partición después. Y eso **suena** a fuga de datos, pero no
lo es: `build_features` solo mira hacia atrás, nunca hacia adelante. Que la
partición venga después es seguro *exactamente porque* eso es cierto.

Que sea cierto no se supone: se testea. El primer test de la Task 10 es lo que
sostiene todo este orden, y si algún día falla, todos los números del repo
dejan de valer.

```python
"""Derived variables, every one of them computed from strictly earlier hours.

The whole file is one rule: a feature for hour H may read H-1 and older, never
H itself. Break it and nothing errors — the scores simply come out excellent
and the model is worthless the moment it faces an hour nobody has lived yet.
"""

import holidays
import pandas as pd

CALENDAR = ["hour", "weekday", "is_weekend", "is_holiday"]
LAGS = [1, 24, 168]
WINDOWS = [24, 168]
FEATURE_COLUMNS = (
    CALENDAR + [f"lag_{h}h" for h in LAGS] + [f"roll_{w}h" for w in WINDOWS]
)


def build_features(frame: pd.DataFrame) -> pd.DataFrame:
    frame = frame.sort_values(["zone_id", "pickup_hour"]).reset_index(drop=True)
    us_holidays = holidays.US(years=sorted(frame.pickup_hour.dt.year.unique()))

    hours = frame.pickup_hour
    out = frame.assign(
        hour=hours.dt.hour.astype("int16"),
        weekday=hours.dt.weekday.astype("int16"),
        is_weekend=(hours.dt.weekday >= 5).astype("int8"),
        is_holiday=hours.dt.date.isin(us_holidays).astype("int8"),
    )

    by_zone = out.groupby("zone_id", sort=False)["trips"]
    for lag in LAGS:
        out[f"lag_{lag}h"] = by_zone.shift(lag)
    for window in WINDOWS:
        # shift(1) first: the window ends at H-1, so hour H is never inside it.
        out[f"roll_{window}h"] = by_zone.transform(
            lambda s, w=window: s.shift(1).rolling(w, min_periods=1).mean()
        )

    # The first week of every zone has no 168h history. Dropping is honest;
    # filling would invent a past that did not happen.
    return out.dropna(subset=FEATURE_COLUMNS).reset_index(drop=True)
```

Fíjate en dos detalles que parecen menores y no lo son:

- `s.shift(1).rolling(w)` y no `s.rolling(w)`. Sin el `shift(1)`, la media móvil
  incluye la propia hora H. Es un carácter de diferencia y es la fuga entera.
- `lambda s, w=window:` con el valor por defecto. Sin él, el cierre captura la
  variable del bucle y las dos ventanas salen iguales — un fallo clásico de
  Python que aquí no daría error, solo números equivocados.

```python
"""Models that answer to the same fit/predict/name as the baselines.

They receive frames that already carry the feature columns, because features
are built once over the whole timeline and the split happens afterwards.
"""

import numpy as np
import pandas as pd
from sklearn.compose import ColumnTransformer
from sklearn.linear_model import Ridge
from sklearn.pipeline import Pipeline
from sklearn.preprocessing import OneHotEncoder, StandardScaler

from src.features import FEATURE_COLUMNS

CATEGORICAL = ["zone_id", "hour", "weekday"]
NUMERIC = [c for c in FEATURE_COLUMNS if c.startswith(("lag_", "roll_"))] + [
    "is_weekend",
    "is_holiday",
]


class LinearModel:
    """Ridge over one-hot zone/hour/weekday plus the lag and rolling columns.

    Linear first on purpose: the coefficients can be read, so when a number
    looks wrong you can ask what the model believes instead of guessing.
    """

    name = "ridge on lags + calendar"

    def __init__(self, alpha: float = 1.0):
        self.pipeline = Pipeline(
            [
                (
                    "encode",
                    ColumnTransformer(
                        [
                            ("cat", OneHotEncoder(handle_unknown="ignore"), CATEGORICAL),
                            ("num", StandardScaler(), NUMERIC),
                        ]
                    ),
                ),
                ("ridge", Ridge(alpha=alpha)),
            ]
        )

    def fit(self, train: pd.DataFrame) -> "LinearModel":
        self.pipeline.fit(train[CATEGORICAL + NUMERIC], train.trips)
        return self

    def predict(self, frame: pd.DataFrame) -> np.ndarray:
        # Demand cannot be negative; a linear model does not know that.
        return np.clip(self.pipeline.predict(frame[CATEGORICAL + NUMERIC]), 0, None)
```

Por qué cada pieza:

- **`OneHotEncoder` para zona, hora y día**: para un modelo lineal, la zona 200
  no es "el doble" de la zona 100. Tratarlas como números le haría aprender una
  relación que no existe.
- **`handle_unknown="ignore"`**: si aparece una zona nueva, el modelo responde
  en vez de reventar en producción.
- **`StandardScaler` en las numéricas**: Ridge penaliza los coeficientes
  grandes, así que si `roll_168h` va en cientos y `is_holiday` en 0/1, la
  penalización castiga a una y no a la otra por pura escala.
- **`np.clip(..., 0, None)`**: la demanda no es negativa.

```python
class BoostedTrees:
    """LightGBM over the same columns, with the categoricals left as categories.

    Trees do not need one-hot: splitting on a category directly is both faster
    and better here. They also do not need scaling — a tree only ever asks
    whether a value is above or below a threshold, so units are irrelevant.
    """

    name = "lightgbm on lags + calendar"

    def __init__(self, **params):
        import lightgbm as lgb

        self.lgb = lgb
        self.params = {
            "objective": "regression_l1",  # matches MAE, the headline metric
            "n_estimators": 400,
            "learning_rate": 0.05,
            "num_leaves": 63,
            "verbose": -1,
            "random_state": 0,
            **params,
        }

    def _frame(self, frame: pd.DataFrame) -> pd.DataFrame:
        columns = frame[CATEGORICAL + NUMERIC].copy()
        for column in CATEGORICAL:
            columns[column] = columns[column].astype("category")
        return columns

    def fit(self, train: pd.DataFrame) -> "BoostedTrees":
        self.model = self.lgb.LGBMRegressor(**self.params)
        self.model.fit(self._frame(train), train.trips)
        return self

    def predict(self, frame: pd.DataFrame) -> np.ndarray:
        return np.clip(self.model.predict(self._frame(frame)), 0, None)
```

`objective="regression_l1"` no es un detalle: le dice al modelo que optimice el
error absoluto, que es la métrica del marcador. Con el `l2` por defecto
optimizaría el error cuadrático, se obsesionaría con las horas punta raras, y
saldría peor **en la métrica que estás publicando**.
