## El problema que esta vuelta existe para resolver

Un modelo en un notebook no es un sistema. Al servirlo aparecen tres cosas que
no existían antes:

1. **El artefacto.** El modelo entrenado tiene que sobrevivir al proceso que lo
   entrenó, y hay que saber *cuál* es. Un modelo sin versión es un modelo que no
   puedes retirar cuando falle.
2. **La historia.** Para predecir la hora H en la zona Z necesitas su
   `lag_168h`, es decir lo que pasó una semana antes. El servicio tiene que
   llevar esa historia encima.
3. **El desajuste entrenar/servir.** Si el cálculo de variables se duplica —una
   versión en el entrenamiento, otra en el servicio— acaban divergiendo. El
   modelo va perfecto en local y miente en producción, y nadie se entera porque
   nada falla.

El punto 3 es el importante, y la defensa está decidida desde la vuelta 2: hay
**un solo `features.py`** y el servicio lo importa. No hay una segunda copia de
esa aritmética en ningún sitio.

```python
"""Train on everything up to the validation month and save one artefact.

The artefact carries the model, a version string, and the tail of history the
service needs to compute lags. It does not carry the full dataset: two weeks
per zone is all the longest lag can reach.
"""

from datetime import datetime, timezone
from pathlib import Path

import joblib
import pandas as pd

from src.config import ROOT
from src.evaluate import load_featured
from src.metrics import mae, wape

ARTIFACTS = ROOT / "artifacts"
# 168h lag + 168h rolling window = 336 hours of history, per zone.
HISTORY_HOURS = 336


def recent_history(frame: pd.DataFrame, hours: int = HISTORY_HOURS) -> pd.DataFrame:
    """The tail each zone needs so the service can build its own features."""
    return (
        frame.sort_values("pickup_hour")
        .groupby("zone_id", sort=False)
        .tail(hours)
        .sort_values(["zone_id", "pickup_hour"])
        .reset_index(drop=True)
    )


def train_and_save(model=None) -> Path:
    if model is None:
        # Whatever won lap 2. Change this one line to switch what is served.
        from src.baselines import ZoneHourWeekdayMean

        model = ZoneHourWeekdayMean()

    train, val = load_featured()
    fitted = model.fit(train)

    prediction = fitted.predict(val)
    metrics = {
        "mae": round(mae(val.trips.to_numpy(), prediction), 3),
        "wape": round(wape(val.trips.to_numpy(), prediction), 4),
    }

    trained_at = datetime.now(timezone.utc)
    bundle = {
        "model": fitted,
        "version": f"{model.name}@{trained_at:%Y%m%dT%H%M%SZ}",
        # Only the raw columns: the service rebuilds features from them with
        # the same code that trained the model.
        "history": recent_history(
            pd.concat([train, val])[["pickup_hour", "zone_id", "trips"]]
        ),
        "trained_at": trained_at.isoformat(),
        "metrics": metrics,
    }

    ARTIFACTS.mkdir(exist_ok=True)
    target = ARTIFACTS / "model.joblib"
    joblib.dump(bundle, target, compress=3)
    print(f"{bundle['version']}  MAE {metrics['mae']}  "
          f"{target.stat().st_size / 1e6:.1f} MB")
    return target


if __name__ == "__main__":
    train_and_save()
```

```python
"""Serve predictions from the artefact that `make train` produced.

Features are rebuilt here by importing src.features — the same module that
trained the model. Two copies of that arithmetic is exactly how a model comes
out fine locally and wrong live, with nothing reporting the difference.
"""

from contextlib import asynccontextmanager
from datetime import datetime

import joblib
import pandas as pd
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel

from src.config import ROOT
from src.features import build_features

MODEL_PATH = ROOT / "artifacts" / "model.joblib"

state: dict = {}


class Prediction(BaseModel):
    zone_id: int
    hour: datetime
    predicted_trips: float
    model_version: str


@asynccontextmanager
async def lifespan(app: FastAPI):
    # Loaded once at startup, not per request: unpickling on every call is the
    # difference between 3 ms and 300 ms.
    if MODEL_PATH.exists():
        bundle = joblib.load(MODEL_PATH)
        state.update(
            model=bundle["model"],
            version=bundle["version"],
            history=bundle["history"],
        )
    yield
    state.clear()


app = FastAPI(title="NYC taxi demand forecast", lifespan=lifespan)


@app.get("/health")
def health() -> dict:
    return {"ok": "model" in state, "version": state.get("version")}


@app.get("/predict", response_model=Prediction)
def predict(zone_id: int, hour: datetime) -> Prediction:
    if "model" not in state:
        raise HTTPException(503, "no model loaded; run `make train` first")

    history = state["history"]
    rows = history[history.zone_id == zone_id]
    if rows.empty:
        raise HTTPException(404, f"zone {zone_id} unknown")

    target = pd.DataFrame(
        [{"pickup_hour": pd.Timestamp(hour), "zone_id": zone_id, "trips": 0}]
    )
    featured = build_features(pd.concat([rows, target], ignore_index=True))
    row = featured[featured.pickup_hour == pd.Timestamp(hour)]
    if row.empty:
        raise HTTPException(422, "not enough history for that hour")

    return Prediction(
        zone_id=zone_id,
        hour=hour,
        predicted_trips=float(state["model"].predict(row)[0]),
        model_version=state["version"],
    )
```

Fíjate en la línea `trips: 0` de `target`. Es una columna de relleno: la hora
que quieres predecir todavía no tiene viajes. Y es seguro ponerla a cero
**porque `build_features` no la mira** — es la propiedad que testeaste en la
Task 10. Si esa propiedad no se cumpliera, esta línea sería un cero
contaminando la predicción.

```go
// A thin gateway in front of the Python service: it terminates the public
// request, forwards to the model, and caches identical questions for a minute.
// The model service never faces the internet directly.
package main

import (
	"fmt"
	"io"
	"log"
	"net/http"
	"os"
	"sync"
	"time"
)

type entry struct {
	body    []byte
	expires time.Time
}

type cache struct {
	mu      sync.RWMutex
	entries map[string]entry
	ttl     time.Duration
}

func newCache(ttl time.Duration) *cache {
	return &cache{entries: map[string]entry{}, ttl: ttl}
}

func (c *cache) get(key string) ([]byte, bool) {
	c.mu.RLock()
	defer c.mu.RUnlock()
	e, ok := c.entries[key]
	if !ok || time.Now().After(e.expires) {
		return nil, false
	}
	return e.body, true
}

func (c *cache) put(key string, body []byte) {
	c.mu.Lock()
	defer c.mu.Unlock()
	c.entries[key] = entry{body: body, expires: time.Now().Add(c.ttl)}
}

func handler(upstream string, store *cache, client *http.Client) http.HandlerFunc {
	return func(w http.ResponseWriter, r *http.Request) {
		key := r.URL.RawQuery
		if body, ok := store.get(key); ok {
			w.Header().Set("Content-Type", "application/json")
			w.Header().Set("X-Cache", "hit")
			w.Write(body)
			return
		}

		resp, err := client.Get(upstream + "/predict?" + r.URL.RawQuery)
		if err != nil {
			w.Header().Set("Content-Type", "application/json")
			http.Error(w, fmt.Sprintf(`{"detail":"model unreachable: %v"}`, err),
				http.StatusBadGateway)
			return
		}
		defer resp.Body.Close()

		body, err := io.ReadAll(resp.Body)
		if err != nil {
			http.Error(w, `{"detail":"could not read model response"}`,
				http.StatusBadGateway)
			return
		}

		// Only successes are cached. Caching a 404 would keep answering it
		// after the zone became known.
		if resp.StatusCode == http.StatusOK {
			store.put(key, body)
		}
		w.Header().Set("Content-Type", "application/json")
		w.Header().Set("X-Cache", "miss")
		w.WriteHeader(resp.StatusCode)
		w.Write(body)
	}
}

func main() {
	upstream := os.Getenv("MODEL_URL")
	if upstream == "" {
		upstream = "http://127.0.0.1:8000"
	}
	port := os.Getenv("PORT")
	if port == "" {
		port = "8080"
	}

	client := &http.Client{Timeout: 5 * time.Second}
	http.HandleFunc("/predict", handler(upstream, newCache(time.Minute), client))
	http.HandleFunc("/health", func(w http.ResponseWriter, r *http.Request) {
		w.Header().Set("Content-Type", "application/json")
		w.Write([]byte(`{"gateway":"ok"}`))
	})

	log.Printf("gateway on :%s forwarding to %s", port, upstream)
	log.Fatal(http.ListenAndServe(":"+port, nil))
}
```

`bog` es Bogotá, la región más cercana. Contesta **no** a añadir Postgres y a
añadir Redis.

- [ ] **Paso 3: Ajustar `fly.toml`**

```toml
app = "duvan-demand-forecast"
primary_region = "bog"

[build]
  dockerfile = "Dockerfile"

[http_service]
  internal_port = 8000
  force_https = true
  auto_stop_machines = "stop"
  auto_start_machines = true
  # Zero machines when nobody is asking. It costs nothing while idle and the
  # first request after a pause pays a few seconds of cold start.
  min_machines_running = 0

[[vm]]
  memory = "512mb"
  cpu_kind = "shared"
  cpus = 1
```

Empieza desplegando **solo el servicio de Python**. La pasarela añade una
segunda app y complica el primer despliegue; una vez esto responda, la añades.
