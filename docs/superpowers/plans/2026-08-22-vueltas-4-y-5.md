- [ ] **Paso 3: Escribir `src/backtest.py`**

```python
"""Walk-forward backtest and a drift signal.

One score on one validation month says how the model did in November. It says
nothing about whether it will still work in March. Walking the model forward a
month at a time is the only way to see the slope.
"""

import numpy as np
import pandas as pd

from src.config import REPORTS_DIR
from src.metrics import mae, wape


def walk_forward(model_factory, featured: pd.DataFrame, min_train_months: int = 4):
    """Train on everything before month M, score on M, step forward, repeat.

    model_factory is called fresh for each fold: reusing a fitted model would
    carry knowledge of later months backwards, which is the leak this whole
    project is built to avoid.
    """
    months = sorted(featured.pickup_hour.dt.strftime("%Y-%m").unique())
    month_of = featured.pickup_hour.dt.strftime("%Y-%m")

    rows = []
    for i in range(min_train_months, len(months)):
        target = months[i]
        train = featured[month_of < target]
        test = featured[month_of == target]

        prediction = model_factory().fit(train).predict(test)
        rows.append(
            {
                "month": target,
                "train_months": i,
                "mae": round(mae(test.trips.to_numpy(), prediction), 3),
                "wape": round(wape(test.trips.to_numpy(), prediction), 4),
            }
        )
    return pd.DataFrame(rows)


def psi(expected: np.ndarray, actual: np.ndarray, buckets: int = 10) -> float:
    """Population Stability Index: how far a distribution has moved.

    Rule of thumb from credit scoring, and good enough here: under 0.1 nothing
    happened, 0.1-0.25 is worth a look, over 0.25 the world changed and the
    model is answering a question nobody asked any more.
    """
    edges = np.quantile(expected, np.linspace(0, 1, buckets + 1))
    edges[0], edges[-1] = -np.inf, np.inf
    e = np.histogram(expected, bins=edges)[0] / len(expected)
    a = np.histogram(actual, bins=edges)[0] / len(actual)
    # Nudge empty buckets off zero so the log stays finite.
    e, a = np.clip(e, 1e-6, None), np.clip(a, 1e-6, None)
    return float(np.sum((a - e) * np.log(a / e)))


def main() -> None:
    from src.baselines import ZoneHourWeekdayMean
    from src.features import build_features
    from src.ingest import load_all
    from src.models import BoostedTrees, LinearModel

    featured = build_features(load_all())

    sections = []
    for factory in (ZoneHourWeekdayMean, LinearModel, BoostedTrees):
        table = walk_forward(factory, featured)
        trend = table.mae.iloc[-1] - table.mae.iloc[0]
        sections.append(
            f"## {factory().name}\n\n{table.to_markdown(index=False)}\n\n"
            f"MAE {'sube' if trend > 0 else 'baja'} {abs(trend):.3f} entre el "
            f"primer mes evaluado y el último.\n"
        )

    REPORTS_DIR.mkdir(parents=True, exist_ok=True)
    target = REPORTS_DIR / "backtest.md"
    target.write_text("# Backtest walk-forward\n\n" + "\n".join(sections))
    print(f"written to {target}")


if __name__ == "__main__":
    main()
```

```python
"""Compare the distribution the model was trained on against the recent one.

Drift is not an error. It is the world changing while the model keeps answering
the old question, and it does not announce itself: the service stays up, the
latency stays flat, and the predictions quietly stop being right.
"""

import pandas as pd

from src.backtest import psi
from src.config import REPORTS_DIR
from src.features import FEATURE_COLUMNS


def _verdict(value: float) -> str:
    if value < 0.1:
        return "estable"
    if value < 0.25:
        return "vigilar"
    return "ha cambiado"


def drift_report(
    featured: pd.DataFrame, reference: list[str], recent: list[str]
) -> pd.DataFrame:
    month_of = featured.pickup_hour.dt.strftime("%Y-%m")
    before = featured[month_of.isin(reference)]
    after = featured[month_of.isin(recent)]

    rows = []
    for column in FEATURE_COLUMNS:
        value = psi(before[column].to_numpy(), after[column].to_numpy())
        rows.append(
            {"feature": column, "psi": round(value, 4), "verdict": _verdict(value)}
        )
    return pd.DataFrame(rows).sort_values("psi", ascending=False).reset_index(drop=True)


def main() -> None:
    from src.config import MONTHS
    from src.features import build_features
    from src.ingest import load_all

    featured = build_features(load_all())
    reference, recent = MONTHS[:3], MONTHS[-3:]
    report = drift_report(featured, reference, recent)

    REPORTS_DIR.mkdir(parents=True, exist_ok=True)
    target = REPORTS_DIR / "drift.md"
    target.write_text(
        "# Deriva\n\n"
        f"PSI entre {reference[0]}-{reference[-1]} (referencia) y "
        f"{recent[0]}-{recent[-1]} (reciente).\n\n"
        "Menos de 0,1 estable · 0,1-0,25 vigilar · más de 0,25 ha cambiado.\n\n"
        f"{report.to_markdown(index=False)}\n"
    )
    print(report.to_string(index=False))
    print(f"\nwritten to {target}")


if __name__ == "__main__":
    main()
```

```yaml
  schedule:
    # The TLC publishes a month roughly two weeks after it ends. The 20th is a

jobs:
  retrain:
    runs-on: ubuntu-latest
    timeout-minutes: 45
    steps:
      - uses: actions/checkout@v4

      - uses: astral-sh/setup-uv@v5
        with:
          enable-cache: true

      - name: Install
        run: uv sync --frozen

      - name: Download and ingest
        run: |
          uv run python -m src.download
          uv run python -m src.ingest

      - name: Score, backtest and check for drift
        run: |
          uv run python -m src.evaluate
          uv run python -m src.backtest
          uv run python -m src.drift

      - name: Fail if the world moved too far
        run: |
          uv run python -c "

          from src.config import MONTHS
          from src.drift import drift_report
          from src.features import build_features
          from src.ingest import load_all
          report = drift_report(build_features(load_all()), MONTHS[:3], MONTHS[-3:])

              sys.exit(f'drift on {worst.feature}: PSI {worst.psi} — retrain and review')

      - name: Open a pull request with the new numbers
        uses: peter-evans/create-pull-request@v7
        with:
          branch: retrain/${{ github.run_number }}
          title: "Retrain: refreshed scoreboard, backtest and drift"
```

```python
class NeuralNet:
    """A small MLP over the same columns as the ridge.

    Here to answer one question with a number: does a neural network beat
    gradient boosting on tabular data like this? Usually it does not, and
    publishing that is the point.
    """

    name = "mlp on lags + calendar"

    def __init__(self, max_iter: int = 60):
        from sklearn.neural_network import MLPRegressor

        self.pipeline = Pipeline(
            [
                (
                    "encode",
                    ColumnTransformer(
                        [
                            ("cat", OneHotEncoder(handle_unknown="ignore"), CATEGORICAL),
                            ("num", StandardScaler(), NUMERIC),
                        ]
                    ),
                ),
                (
                    "mlp",
                    MLPRegressor(
                        hidden_layer_sizes=(64, 32),
                        max_iter=max_iter,
                        # Stop when the held-out slice stops improving, rather
                        # than running the full budget into overfitting.
                        early_stopping=True,
                        random_state=0,
                    ),
                ),
            ]
        )

    def fit(self, train: pd.DataFrame) -> "NeuralNet":
        self.pipeline.fit(train[CATEGORICAL + NUMERIC], train.trips)
        return self

    def predict(self, frame: pd.DataFrame) -> np.ndarray:
        return np.clip(self.pipeline.predict(frame[CATEGORICAL + NUMERIC]), 0, None)
```
